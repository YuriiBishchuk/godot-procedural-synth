Install via NuGet or clone the addon into your project:

# Install NuGet package into your Godot C# project dotnet add package GodotProceduralSynth // In your Godot Node script (e.g. SynthPlayer.cs) using Godot; using GodotProceduralSynth; public partial class SynthPlayer : Node { private AudioStreamGeneratorPlayback _playback; private UniversalSynthEngine _engine; public override void _Ready() { var player = GetNode<AudioStreamPlayer>("AudioStreamPlayer"); var generator = new AudioStreamGenerator { MixRate = 44100, BufferLength = 0.1f }; player.Stream = generator; player.Play(); _playback = (AudioStreamGeneratorPlayback)player.GetStreamPlayback(); _engine = new UniversalSynthEngine(); } public override void _Process(double delta) { int available = _playback.GetFramesAvailable(); if (available > 0) { var left = new float[available]; var right = new float[available]; // Generate adaptive audio in real time driven by gameplay tension! _engine.GenerateStreamBlock(left, right, available, new AdaptiveAudioContext { Tension = 0.5f, ActivityRate = 120.0, IntensityLevel = 5 }); for (int i = 0; i < available; i++) _playback.PushFrame(new Vector2(left[i], right[i])); } } }
